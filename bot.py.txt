import os
import subprocess
import sys

def main():
    print("Starting Hermes Gateway for Telegram...")
    # Kiểm tra biến môi trường Token Telegram
    token = os.environ.get("TELEGRAM_BOT_TOKEN")
    if token:
        print("Telegram Bot Token loaded successfully.")
    
    # Khởi chạy lệnh gateway của Hermes
    try:
        subprocess.run(["hermes", "gateway", "start"], check=True)
    except Exception as e:
        print(f"Error running hermes gateway: {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()
